// ESP32_SmartFarm_Firmware.ino 
#include <WiFi.h>
#include <PubSubClient.h>
#define SOIL_ADC 34
#define PUMP_RELAY 16

void loop() {
  int moisture = analogRead(SOIL_ADC);
  if (moisture < 400) {
    digitalWrite(PUMP_RELAY, HIGH);
    publishMQTT("farm/pump", "ON");
  }
}
IC: ESP32-D0WDQ6 ADC: 12-Bit 0-3.3V
